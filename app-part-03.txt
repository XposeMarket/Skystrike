function updateFlight(dt) {
  const cfg=aircraftConfigs[state.currentAircraft];

  // Mobile flight-stick controls: horizontal input commands a bank AND a coordinated turn.
  // Right stick = right bank/right turn, left stick = left bank/left turn.
  // Vertical input keeps traditional flight-stick behavior: pull down to pitch up.
  const maxRoll=1.08, maxPitch=.82;
  const rollBlend=1-Math.exp(-dt*5.4), pitchBlend=1-Math.exp(-dt*4.2);
  const targetRoll=-state.stickX*maxRoll;
  const targetPitch=state.stickY*maxPitch;
  state.roll=THREE.MathUtils.lerp(state.roll,targetRoll,rollBlend);
  state.pitch=THREE.MathUtils.lerp(state.pitch,targetPitch,pitchBlend);
  state.pitch=THREE.MathUtils.clamp(state.pitch,-1.2,1.2);
  state.roll=THREE.MathUtils.clamp(state.roll,-1.45,1.45);

  // In this coordinate system decreasing yawAngle turns the heading right.
  // Negating rudder input fixes the old left/right reversal. Bank coupling makes
  // the aircraft actually carve through a turn instead of merely rolling in place.
  const rudderYawRate=state.mode==='space'?.92:1.05;
  const bankYawRate=state.mode==='space'?.62:.78;
  const directTurnRate=state.mode==='space'?.16:.24;
  const speedTurnFactor=state.mode==='earth'?THREE.MathUtils.clamp(180/Math.max(state.speed,80),.55,1.35):1;
  const turnRate=(-state.yaw*rudderYawRate + Math.sin(state.roll)*bankYawRate - state.stickX*directTurnRate)*speedTurnFactor;
  state.yawAngle += turnRate*dt;

  if(state.mode==='earth'){
    const max = cfg.maxSpeed*(state.boost?1.16:1);
    const target=Math.max(22,state.throttle*max);
    state.speed += (target-state.speed)*Math.min(1,dt*(state.speed<target?.7:1.4));
    flightEuler.set(state.pitch,state.yawAngle,state.roll,'YXZ');flightQuat.setFromEuler(flightEuler);
    const forward=tmpV.set(0,0,-1).applyQuaternion(flightQuat).normalize();
    const lift=Math.max(0,(state.speed-cfg.minSpeed)/(cfg.maxSpeed-cfg.minSpeed));
    const previousX=flightPos.x,previousZ=flightPos.z;
    flightPos.addScaledVector(forward,state.speed*dt);
    const buildingHit=buildingCollisionAt(flightPos.x,flightPos.z,flightPos.y);
    if(buildingHit){
      flightPos.x=previousX;flightPos.z=previousZ;state.speed=Math.max(22,state.speed*.35);state.roll*=.35;
      if(performance.now()-state.lastBuildingHitAt>2200){state.lastBuildingHitAt=performance.now();toast('BUILDING HIT - PULL UP');}
    }
    flightPos.y += (lift*.35 - (1-lift)*1.8)*dt;
    state.currentGeo=geoFromFlight();
    state.terrainBase=terrainHeightAt(flightPos.x,flightPos.z);
    const minY=state.terrainBase+12;
    if(flightPos.y<minY){flightPos.y=minY;state.pitch=Math.max(state.pitch,.05);state.speed*=.985;}

    // Automatic orbital handoff: keep climbing hard in an atmospheric aircraft and
    // the game swaps to the Shuttle before the service ceiling, preserving the flight.
    if(!cfg.space && flightPos.y>cfg.ceiling*.78 && state.throttle>.72 && state.pitch>.10){
      toast('AUTO ORBIT Â· SWITCHING TO SPACE SHUTTLE');
      setAircraft('shuttle',true);
    }
    if(!cfg.space&&flightPos.y>cfg.ceiling){flightPos.y=cfg.ceiling;state.pitch=Math.min(state.pitch,0);state.speed*=.992;}

    const atmoFade=THREE.MathUtils.clamp(1-flightPos.y/85000,0,1);if(skyMaterial)skyMaterial.uniforms.fade.value=atmoFade;scene.fog.density=.000018*Math.pow(atmoFade,1.8);
    state.terrainClock+=dt;if(state.terrainClock>.65){state.terrainClock=0;refreshTerrainTiles(false);rebaseEarth();if(state.buildings)loadBuildings(state.currentGeo.lat,state.currentGeo.lon);}

    // No ORBIT button required: once the Shuttle reaches the upper atmosphere,
    // transition into the orbital coordinate system automatically.
    if(cfg.space && flightPos.y>45000){toast('SPACE REACHED Â· ENTERING ORBIT');enterSpace('Earth');return;}

    state.altitude=flightPos.y;
    state.heading=(THREE.MathUtils.radToDeg(-state.yawAngle)%360+360)%360;
  }else{
    const accel=(0.25+state.throttle*1.6)*(state.boost?3.5:1);
    state.spaceVelocity=THREE.MathUtils.clamp(state.spaceVelocity+accel*dt*.05,.15,9);
    flightEuler.set(state.pitch,state.yawAngle,state.roll,'YXZ');flightQuat.setFromEuler(flightEuler);
    const forward=tmpV.set(0,0,-1).applyQuaternion(flightQuat).normalize();
    flightPos.addScaledVector(forward,state.spaceVelocity*1100*dt);
    state.speed=state.spaceVelocity*3200;
    const targetBody=solarBodies.get(state.spaceTarget);
    if(targetBody){const p=targetBody.mesh.getWorldPosition(tmpV2);state.altitude=flightPos.distanceTo(p)*1000;}
    state.heading=(THREE.MathUtils.radToDeg(-state.yawAngle)%360+360)%360;
  }
  aircraftRoot.position.copy(flightPos);aircraftRoot.quaternion.copy(flightQuat);
}

function updateCamera(dt) {
  const forward=tmpV.set(0,0,-1).applyQuaternion(flightQuat).normalize();
  const up=tmpV2.set(0,1,0).applyQuaternion(flightQuat).normalize();
  let desired;
  if(state.cameraMode===0){desired=flightPos.clone().addScaledVector(forward, state.mode==='space' ? -200 : -35).addScaledVector(up,state.mode==='space'?70:12);}
  else if(state.cameraMode===1){desired=flightPos.clone().addScaledVector(forward,5).addScaledVector(up,2.2);}
  else {desired=flightPos.clone().add(new THREE.Vector3(0,state.mode==='space'?350:55,state.mode==='space'?500:80));}
  if(state.cameraMode===1){camera.position.lerp(desired,1-Math.pow(.001,dt));camTarget.copy(flightPos).addScaledVector(forward,100);}else{camera.position.lerp(desired,1-Math.pow(.02,dt));camTarget.copy(flightPos).addScaledVector(forward,state.mode==='space'?500:120);}
  camera.up.lerp(up, state.cameraMode===1?.15:.04).normalize();
  camera.lookAt(camTarget);
}
