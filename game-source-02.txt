function addCylinder(parent,r1,r2,len,material,pos,rot={x:Math.PI/2,y:0,z:0},segments=16){const m=new THREE.Mesh(new THREE.CylinderGeometry(r1,r2,len,segments),material);m.position.copy(pos);m.rotation.set(rot.x,rot.y,rot.z);m.castShadow=true;m.receiveShadow=true;parent.add(m);return m}
function addBox(parent,size,material,pos,rot={x:0,y:0,z:0}){const m=new THREE.Mesh(new THREE.BoxGeometry(size.x,size.y,size.z),material);m.position.copy(pos);m.rotation.set(rot.x,rot.y,rot.z);m.castShadow=true;m.receiveShadow=true;parent.add(m);return m}

    const cockpit=new THREE.Mesh(new THREE.TorusGeometry(.64,.11,8,20,Math.PI),darkMat);cockpit.rotation.x=Math.PI/2;cockpit.position.set(0,.7,.2);g.add(cockpit);const pilot=new THREE.Mesh(new THREE.SphereGeometry(.28,12,8),new THREE.MeshStandardMaterial({color:0x94735c}));pilot.position.set(0,.82,.05);g.add(pilot);

function createPlayer(cfg){
  const root=new THREE.Group(),model=createPlaneModel(cfg,false);root.add(model);root.position.set(0,surfaceHeight(0,-470)+cfg.gearHeight,-470);
  const muzzleFlash=new THREE.Mesh(new THREE.ConeGeometry(.22,.95,6),new THREE.MeshBasicMaterial({color:0xffd27a,transparent:true,opacity:.9,blending:THREE.AdditiveBlending,depthWrite:false}));muzzleFlash.rotation.x=Math.PI/2;muzzleFlash.position.set(0,0,5.2);muzzleFlash.visible=false;root.add(muzzleFlash);scene.add(root);
  return {root,model,muzzleFlash,muzzleFlashUntil:0,cfg,speed:0,throttle:0,health:100,ammo:cfg.gun.ammo,reloading:false,reloadEnd:0,lastShot:0,grounded:true,dead:false,afterburner:false};
}

function spawnEnemies(){
  enemies.forEach(e=>scene.remove(e.root));enemies=[];
  const cfg=player.cfg;for(let i=0;i<4;i++)spawnEnemy(cfg,i,true);
}
function spawnEnemy(cfg,index,initial=false){
  const root=new THREE.Group(),model=createPlaneModel(cfg,true);root.add(model);const ang=Math.random()*Math.PI*2,dist=initial?700+Math.random()*750:450+Math.random()*700;root.position.copy(player.root.position).add(new THREE.Vector3(Math.cos(ang)*dist,220+Math.random()*300,Math.sin(ang)*dist));root.quaternion.setFromUnitVectors(FORWARD,new THREE.Vector3(-Math.sin(ang),.05,Math.cos(ang)).normalize());scene.add(root);
  const enemy={root,model,cfg,speed:cfg.maxSpeed*(.54+Math.random()*.18),health:70+(cfg.id==='jet'?30:0),lastShot:Math.random(),dead:false,respawnAt:0,index,offset:Math.random()*100,difficulty};enemies.push(enemy);return enemy;
}

function resetGame(){
  bullets.forEach(b=>scene.remove(b.mesh));effects.forEach(e=>scene.remove(e.obj));bullets=[];effects=[];enemies.forEach(e=>scene.remove(e.root));enemies=[];if(player)scene.remove(player.root);
  const cfg=AIRCRAFT[selectedAircraft];player=createPlayer(cfg);difficulty=parseFloat(dom.difficulty.value);kills=0;gameTime=0;verticalSpeed=0;lastPlayerY=player.root.position.y;viewMode='chase';hudElapsed=radarElapsed=shadowElapsed=0;perfElapsed=perfFrames=perfCooldown=0;easterEgg={triggered:false,active:null,time:0};input.steerX=0;input.steerY=0;spawnEnemies();
  dom.aircraft.textContent=`${cfg.name} • ${cfg.gun.name}`;dom.view.textContent='CHASE';dom.cockpit.classList.add('hidden');player.model.visible=true;updateHUD(0);
  const runwayTop=surfaceHeight(0,-470);camera.position.set(0,runwayTop+8,-490);camera.lookAt(0,runwayTop+2,-450);audio.gameMusic();audio.startEngine(cfg);showBanner('FALCON RIDGE • CLEARED FOR TAKEOFF',2.6);
}

function startGame(){
  audio.ensure();audio.click();if(!renderer)initThree();resetGame();running=true;paused=false;dom.menu.classList.remove('visible');dom.pause.classList.remove('visible');dom.hud.classList.remove('hidden');clock.start();renderer.domElement.requestPointerLock();

function returnToMenu(){

function updatePlayer(dt){
  const p=player,cfg=p.cfg;if(p.dead)return;
  if(keys.KeyW)p.throttle=clamp(p.throttle+dt*.32,0,1);if(keys.KeyS)p.throttle=clamp(p.throttle-dt*.38,0,1);
  const rollInput=(keys.KeyA?-1:0)+(keys.KeyD?1:0),barrel=keys.Space?1:0;
  input.steerX*=Math.pow(.001,dt);input.steerY*=Math.pow(.001,dt);
  const yawInput=clamp(input.steerX,-1,1),pitchInput=clamp(-input.steerY,-1,1);
  const prevY=p.root.position.y;
  if(p.grounded){
    const target=p.throttle*cfg.maxSpeed*.68;p.speed+=clamp(target-p.speed,-cfg.maxSpeed, cfg.maxSpeed)*dt*(cfg.engineResponse*1.7);p.speed=Math.max(0,p.speed-dt*(1-p.throttle)*2.2);
    if(Math.abs(yawInput)>.02)p.root.rotateY(yawInput*.38*dt*(.3+p.speed/cfg.takeoffSpeed));
    if(p.speed>cfg.takeoffSpeed*.72)p.root.rotateX(-pitchInput*cfg.pitchRate*.42*dt);
    p.root.rotation.z=lerp(p.root.rotation.z,0,dt*4);const f=FORWARD.clone().applyQuaternion(p.root.quaternion);p.root.position.addScaledVector(f,p.speed*dt);const ground=surfaceHeight(p.root.position.x,p.root.position.z)+cfg.gearHeight;p.root.position.y=ground;
    const pitchUp=f.y;if(p.speed>cfg.takeoffSpeed&&pitchUp>.035){p.grounded=false;showBanner('AIRBORNE',1.2);}
  } else {
    const controlAuthority=clamp((p.speed-cfg.stallSpeed*.55)/(cfg.maxSpeed*.48),.2,1.18);
    p.root.rotateX(-pitchInput*cfg.pitchRate*controlAuthority*dt);p.root.rotateY(yawInput*cfg.yawRate*controlAuthority*dt);p.root.rotateZ(rollInput*cfg.rollRate*(barrel?1.65:1)*controlAuthority*dt);
    const right=RIGHT.clone().applyQuaternion(p.root.quaternion),bank=Math.asin(clamp(right.y,-1,1));const bankTurn=new THREE.Quaternion().setFromAxisAngle(UP,-bank*cfg.turnRate*controlAuthority*dt);p.root.quaternion.premultiply(bankTurn).normalize();
