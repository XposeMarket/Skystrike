function solarDistance(millionKm) {
  if (millionKm === 0) return 0;
  return 6000 + Math.log10(millionKm + 1) * 16000;
}
function solarRadius(km) {
  if (km > 150000) return 2100;
  return 210 + Math.pow(km / 6371, 0.42) * 430;
}

function loadTexture(url, color = 0xffffff) {
  return new Promise((resolve) => {
    if (!url) return resolve(null);
    texLoader.load(url, t => { t.colorSpace = THREE.SRGBColorSpace; t.anisotropy = Math.min(8, renderer.capabilities.getMaxAnisotropy()); resolve(t); }, undefined, () => resolve(null));
  });
}

    texLoader.load(url, tex => {
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.anisotropy = Math.min(8, renderer.capabilities.getMaxAnisotropy());

      material.color.setHex(0xffffff);

    }, undefined, () => {});
  };
  for (let i=0;i<planetDefs.length;i++) {
    const p=planetDefs[i];
    const pivot = new THREE.Group();
    const orbitR = solarDistance(p.orbit);
    if (orbitR > 0) {
      const pts=jplElements[p.name]?orbitVisualPoints(p.name):Array.from({length:128},(_,j)=>{const a=j/128*Math.PI*2;return new THREE.Vector3(Math.cos(a)*orbitR,0,Math.sin(a)*orbitR)});

      ring.userData.orbitLine = true; solarGroup.add(ring);

    const mat = i===0 ? new THREE.MeshBasicMaterial({ color:p.color }) : new THREE.MeshStandardMaterial({ color:p.color, roughness:.86, metalness:0 });
    streamTexture(p.texture ? textureBase + p.texture : null, mat);
    const mesh = new THREE.Mesh(geom, mat);
    const livePos=planetVisualPosition(p.name);if(livePos)mesh.position.copy(livePos);else mesh.position.x=orbitR;
    mesh.userData.name = p.name;
    pivot.userData = { def:p, mesh, angle: Math.random()*Math.PI*2 };
    pivot.add(mesh);
    solarGroup.add(pivot);
    solarBodies.set(p.name, { pivot, mesh, def:p });
    if (p.ring) {
      const rg = new THREE.RingGeometry(solarRadius(p.radius)*1.22, solarRadius(p.radius)*2.2, 96);
      const rm = new THREE.MeshBasicMaterial({ color:0xf3dfb0, side:THREE.DoubleSide, transparent:true, opacity:.82, depthWrite:false });

  }

function fallbackAircraft() {

async function setAircraft(key, silent=false) {
  if (!aircraftConfigs[key]) return;
  const cfg = aircraftConfigs[key];
  if(state.mode==='space'&&!cfg.space){if(!silent)toast(`${cfg.name} is atmospheric only`);return;}
  state.currentAircraft = key;
  state.speed = Math.min(state.speed || cfg.cruise, cfg.cruise);
  document.querySelectorAll('[data-aircraft]').forEach(el=>el.classList.toggle('active',el.dataset.aircraft===key));
  if (aircraftVisual) { aircraftRoot.remove(aircraftVisual); aircraftVisual = null; }
  const token = ++aircraftLoadToken;
  if (!silent) toast(`Loading ${cfg.name}`);
  loader.load(cfg.model, gltf => {
    if (token !== aircraftLoadToken) return;
    const raw=gltf.scene;
    raw.updateMatrixWorld(true);
    const box=new THREE.Box3().setFromObject(raw),size=box.getSize(new THREE.Vector3()),center=box.getCenter(new THREE.Vector3());
    const wrapper=new THREE.Group();raw.position.sub(center);wrapper.add(raw);
    const largest=Math.max(size.x,size.y,size.z,.001);wrapper.scale.setScalar(cfg.displaySize/largest);wrapper.rotation.set(...cfg.rotation);
    wrapper.traverse(o=>{if(o.isMesh){o.castShadow=false;o.receiveShadow=false;}});
    aircraftVisual=wrapper;aircraftRoot.add(aircraftVisual);
    if (!silent) toast(`${cfg.name} ready`);
  }, undefined, () => {
    if (token !== aircraftLoadToken) return;
    aircraftVisual = fallbackAircraft();
    aircraftRoot.add(aircraftVisual);
    toast('Remote aircraft unavailable · using flight fallback');
  });
}

function tileXY(lat, lon, z) {
  const n = 2 ** z;
  const x = (lon + 180) / 360 * n;
  const latR = lat * Math.PI / 180;
  const y = (1 - Math.asinh(Math.tan(latR)) / Math.PI) / 2 * n;
  return {x,y};
}

function imageData(url) {
  return new Promise((resolve,reject)=>{
    const img=new Image(); img.crossOrigin='anonymous';
    img.onload=()=>{try{const c=document.createElement('canvas');c.width=img.width;c.height=img.height;const cx=c.getContext('2d',{willReadFrequently:true});cx.drawImage(img,0,0);resolve(cx.getImageData(0,0,img.width,img.height));}catch(e){reject(e)}};
    img.onerror=reject; img.src=url;
  });
}

function terrainTileBounds(tx, ty, z) {
  const n = 2 ** z, wrappedX = ((tx % n) + n) % n;
  const west = wrappedX / n * 360 - 180;
  const east = (wrappedX + 1) / n * 360 - 180;
  const north = THREE.MathUtils.radToDeg(Math.atan(Math.sinh(Math.PI * (1 - 2 * ty / n))));
  const south = THREE.MathUtils.radToDeg(Math.atan(Math.sinh(Math.PI * (1 - 2 * (ty + 1) / n))));
  return { south, west, north, east, centerLat: (south + north) * .5, centerLon: (west + east) * .5 };
}

async function createTerrainTile(tx,ty,z,originTile,tileMeters,detail=false) {
  const seg=detail ? 56 : 28;
  const geo=new THREE.PlaneGeometry(tileMeters,tileMeters,seg,seg);
  const n=2**z;
  const serverX=((tx%n)+n)%n;
  const serverY=Math.max(0,Math.min(n-1,ty));
  const bounds=terrainTileBounds(tx,ty,z);
  const sat=`https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/${z}/${serverY}/${serverX}`;
  const ez=Math.min(z,15), shift=Math.max(0,z-15);
  const elevation=`https://s3.amazonaws.com/elevation-tiles-prod/terrarium/${ez}/${serverX >> shift}/${serverY >> shift}.png`;
  const texture=await loadTexture(sat);
  if(texture){
    texture.colorSpace=THREE.SRGBColorSpace;
    texture.anisotropy=Math.min(16,renderer.capabilities.getMaxAnisotropy());
    texture.minFilter=THREE.LinearMipmapLinearFilter;
    texture.magFilter=THREE.LinearFilter;
  }
  const mat=new THREE.MeshStandardMaterial({map:texture,color:texture?0xffffff:0x3a463b,roughness:.96,metalness:0,polygonOffset:detail,polygonOffsetFactor:detail?-1:0,polygonOffsetUnits:detail?-1:0});
  const heights=new Float32Array((seg+1)*(seg+1));
  let sum=0,count=0;
  try {
    const data=await imageData(elevation);
    const pos=geo.attributes.position;
    for(let i=0;i<pos.count;i++){
      const vx=(i%(seg+1))/seg;
      const vy=Math.floor(i/(seg+1))/seg;
      const px=Math.max(0,Math.min(data.width-1,Math.floor(vx*(data.width-1))));
      const py=Math.max(0,Math.min(data.height-1,Math.floor(vy*(data.height-1))));
      const idx=(py*data.width+px)*4;
      const h=data.data[idx]*256+data.data[idx+1]+data.data[idx+2]/256-32768;
      heights[i]=h; pos.setZ(i,h); sum+=h; count++;
    }
    pos.needsUpdate=true; geo.computeVertexNormals();
  } catch { /* flat fallback if elevation provider/CORS is unavailable */ }
  const mesh=new THREE.Mesh(geo,mat);
  mesh.rotation.x=-Math.PI/2;
  const center=localFromGeo(bounds.centerLat,bounds.centerLon);
  mesh.position.x=center.x; mesh.position.z=center.z;
  mesh.userData={tx,ty,z,size:tileMeters,seg,heights,avgHeight:count?sum/count:0,detail};
  return mesh;
}

function disposeObject(o) {
  o.traverse?.(n=>{
    if(n.geometry)n.geometry.dispose?.();
    if(n.material){
      const mats=Array.isArray(n.material)?n.material:[n.material];
      mats.forEach(m=>{if(m.map)m.map.dispose?.();m.dispose?.();});
    }
  });
}

function clearGroup(group) {
  while(group.children.length){const o=group.children[group.children.length-1];group.remove(o);disposeObject(o);}
}
