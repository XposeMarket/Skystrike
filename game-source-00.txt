      <div class="era">${cfg.era}</div><h3>${cfg.name}</h3>
      <p>${cfg.description}</p>

      selectedAircraft = cfg.id;

class AudioSystem {
  constructor() { this.ctx = null; this.master = null; this.musicTimer = null; this.mode = ''; this.engine = []; this.engineGain = null; this.noiseSource = null; this.noiseGain = null; this.noiseBuffer = null; this.lastGunSound = 0; }
  ensure() {
    if (this.ctx) { if (this.ctx.state === 'suspended') this.ctx.resume(); return; }
    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
    this.master = this.ctx.createGain(); this.master.gain.value = .42; this.master.connect(this.ctx.destination);
    this.noiseBuffer = this.ctx.createBuffer(1, this.ctx.sampleRate * 2, this.ctx.sampleRate);
    const noiseData = this.noiseBuffer.getChannelData(0);
    for (let i=0;i<noiseData.length;i++) noiseData[i] = Math.random()*2-1;
  }
  tone(freq, duration=.1, type='sine', gain=.08, detune=0) {
    if (!this.ctx) return;
    const o = this.ctx.createOscillator(), g = this.ctx.createGain();
    o.type = type; o.frequency.value = freq; o.detune.value = detune;
    g.gain.setValueAtTime(gain, this.ctx.currentTime); g.gain.exponentialRampToValueAtTime(.0001, this.ctx.currentTime + duration);
    o.connect(g); g.connect(this.master); o.start(); o.stop(this.ctx.currentTime + duration);
  }
  noise(duration=.12, gain=.09, filterFreq=1400) {
    if (!this.ctx || !this.noiseBuffer) return;
    const src = this.ctx.createBufferSource(), filter = this.ctx.createBiquadFilter(), g = this.ctx.createGain();
    src.buffer = this.noiseBuffer; filter.type='lowpass'; filter.frequency.value=filterFreq;
    const now=this.ctx.currentTime; g.gain.setValueAtTime(gain,now); g.gain.exponentialRampToValueAtTime(.0001,now+duration);
    src.connect(filter); filter.connect(g); g.connect(this.master); src.start(now,Math.random()*1.7,duration); src.stop(now+duration+.02);
  }
  click() { this.ensure(); this.tone(280,.045,'square',.03); }
  menuMusic() {
    this.ensure(); this.stopMusic(); this.mode='menu';
    const notes = [55, 82.41, 61.74, 92.5, 55, 73.42, 65.41, 98]; let i=0;
    const tick = () => { if(this.mode!=='menu')return; this.tone(notes[i%notes.length],1.15,'sawtooth',.026); this.tone(notes[(i+2)%notes.length]*2,.38,'triangle',.018,-9); if(i%4===0)this.noise(.3,.015,350); i++; };
    tick(); this.musicTimer=setInterval(tick,620);
  }
  gameMusic() {
    this.ensure(); this.stopMusic(); this.mode='game';
    const bass=[73.4,73.4,82.4,65.4,73.4,98,82.4,65.4], high=[293.7,329.6,349.2,246.9]; let i=0;
    const tick=()=>{ if(this.mode!=='game')return; this.tone(bass[i%bass.length],.42,'triangle',.014); if(i%2===0)this.tone(high[(i/2)%high.length|0],.2,'square',.006); i++; };
    tick(); this.musicTimer=setInterval(tick,320);
  }
  stopMusic(){ if(this.musicTimer)clearInterval(this.musicTimer); this.musicTimer=null; this.mode=''; }
  startEngine(cfg){
    this.stopEngine(); if(!this.ctx)return;
    this.engineGain=this.ctx.createGain(); this.engineGain.gain.value=.001; this.engineGain.connect(this.master);
    const types=cfg.id==='jet'?['sawtooth','sine']:['sawtooth','square'];
    this.engine=types.map((type,i)=>{const o=this.ctx.createOscillator();o.type=type;o.frequency.value=cfg.engineTone*(i+1);const g=this.ctx.createGain();g.gain.value=i?0.08:.13;o.connect(g);g.connect(this.engineGain);o.start();return o;});
    const buffer=this.ctx.createBuffer(1,this.ctx.sampleRate*2,this.ctx.sampleRate),data=buffer.getChannelData(0);for(let i=0;i<data.length;i++)data[i]=Math.random()*2-1;
    this.noiseSource=this.ctx.createBufferSource();this.noiseSource.buffer=buffer;this.noiseSource.loop=true;
    const filter=this.ctx.createBiquadFilter();filter.type='bandpass';filter.frequency.value=cfg.id==='jet'?900:280;filter.Q.value=.7;
    this.noiseGain=this.ctx.createGain();this.noiseGain.gain.value=.02;this.noiseSource.connect(filter);filter.connect(this.noiseGain);this.noiseGain.connect(this.engineGain);this.noiseSource.start();
  }
  updateEngine(cfg, throttle, speed, afterburner=false){ if(!this.ctx||!this.engineGain)return; const now=this.ctx.currentTime; const base=cfg.engineTone*(.65+throttle*.75+speed/cfg.maxSpeed*.25); this.engine.forEach((o,i)=>o.frequency.setTargetAtTime(base*(i+1),now,.05)); this.engineGain.gain.setTargetAtTime(.22+throttle*.2+(afterburner?.12:0),now,.08); if(this.noiseGain)this.noiseGain.gain.setTargetAtTime(cfg.id==='jet'?(.05+throttle*.08):(.025+throttle*.035),now,.1); }
  stopEngine(){ this.engine.forEach(o=>{try{o.stop()}catch{}});this.engine=[];if(this.noiseSource)try{this.noiseSource.stop()}catch{}this.noiseSource=null;this.engineGain=null; }
  gun(cfg){ this.ensure(); const now=this.ctx?.currentTime||0; if(cfg.id==='jet'&&now-this.lastGunSound<.055)return; this.lastGunSound=now; if(cfg.id==='jet'){this.noise(.055,.075,2300);this.tone(82,.045,'square',.028);}else if(cfg.id==='warhawk'){this.noise(.105,.12,1500);this.tone(105,.08,'sawtooth',.04);}else{this.noise(.07,.07,2100);this.tone(145,.055,'square',.025);} }
  explosion(){this.noise(.8,.32,520);this.tone(46,.8,'sawtooth',.09);}
  hit(){this.noise(.07,.06,3300);this.tone(190,.12,'square',.025);}
  reload(){this.tone(520,.08,'square',.035);setTimeout(()=>this.tone(420,.09,'square',.03),120);}
  strange(){this.tone(41,5,'sawtooth',.05);this.tone(58,4.5,'sine',.06,17);setTimeout(()=>this.noise(1.8,.08,220),600);}
}

    ctx.strokeStyle='rgba(255,255,255,.18)';ctx.lineWidth=1;for(let y=0;y<256;y+=32){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(256,y+Math.random()*2);ctx.stroke()}
  } else if(kind==='terrain'){

  } else if(kind==='metal'){
