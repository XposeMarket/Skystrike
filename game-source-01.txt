function hash2(x,z){const s=Math.sin(x*127.1+z*311.7)*43758.5453123;return s-Math.floor(s)}
function smoothNoise(x,z){const xi=Math.floor(x),zi=Math.floor(z),xf=x-xi,zf=z-zi;const u=xf*xf*(3-2*xf),v=zf*zf*(3-2*zf);return lerp(lerp(hash2(xi,zi),hash2(xi+1,zi),u),lerp(hash2(xi,zi+1),hash2(xi+1,zi+1),u),v)}
function fbm(x,z){let v=0,a=.5,f=1;for(let i=0;i<5;i++){v+=smoothNoise(x*f,z*f)*a;f*=2.03;a*=.49}return v}
function terrainHeight(x,z){
  const n=(fbm(x*.00115,z*.00115)-.5)*195+(fbm(x*.0032+22,z*.0032-17)-.5)*54;
  const ridges=Math.pow(Math.abs(fbm(x*.00055+90,z*.00055+40)-.5)*2,2.4)*480;
  const radial=Math.max(0,(Math.hypot(x,z)-900)/2500)*220;
  let h=12+n+ridges+radial;
  const valley=Math.exp(-(x*x+z*z)/(2*1100*1100));h-=valley*80;
  const runwayMask=Math.max(Math.abs(x)/150,Math.abs(z)/720);
  if(runwayMask<1){const t=THREE.MathUtils.smoothstep(runwayMask,.45,1);h=lerp(RUNWAY_Y-1,h,t)}
  return Math.max(-35,h);
}

function surfaceHeight(x,z){
  // The runway is a raised box, so use its actual top surface instead of the terrain below it.
  if(Math.abs(x)<=46&&Math.abs(z)<=630)return RUNWAY_Y+.6;
  return terrainHeight(x,z);
}
