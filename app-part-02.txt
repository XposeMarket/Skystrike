function unwrapTileX(x,ref,z){const n=2**z;while(x-ref>n/2)x-=n;while(x-ref<-n/2)x+=n;return x;}
function metersPerLonDegree(lat){return 111320*Math.max(.02,Math.cos(lat*Math.PI/180));}
function geoFromFlight(){
  const lat=THREE.MathUtils.clamp(state.earthOrigin.lat-flightPos.z/111320,-85.04,85.04);
  let lon=state.earthOrigin.lon+flightPos.x/metersPerLonDegree(state.earthOrigin.lat);

  return {lat,lon};
}
function localFromGeo(lat,lon){
  let dlon=lon-state.earthOrigin.lon;if(dlon>180)dlon-=360;if(dlon<-180)dlon+=360;
  return {x:dlon*metersPerLonDegree(state.earthOrigin.lat),z:(state.earthOrigin.lat-lat)*111320};
}

function terrainHeightAt(x,z){
  for(const mesh of terrainTileMap.values()){
    const {size,seg,heights}=mesh.userData;
    const u=(x-(mesh.position.x-size/2))/size;
    const v=(z-(mesh.position.z-size/2))/size;
    if(u<0||u>1||v<0||v>1)continue;
    const gx=THREE.MathUtils.clamp(u*seg,0,seg), gy=THREE.MathUtils.clamp(v*seg,0,seg);
    const x0=Math.floor(gx),y0=Math.floor(gy),x1=Math.min(seg,x0+1),y1=Math.min(seg,y0+1);
    const fx=gx-x0,fy=gy-y0,idx=(xx,yy)=>yy*(seg+1)+xx;
    const h00=heights[idx(x0,y0)],h10=heights[idx(x1,y0)],h01=heights[idx(x0,y1)],h11=heights[idx(x1,y1)];
    return THREE.MathUtils.lerp(THREE.MathUtils.lerp(h00,h10,fx),THREE.MathUtils.lerp(h01,h11,fx),fy);
  }
  return state.terrainBase||0;
}

async function refreshTerrainTiles(initial=false){
  if(state.mode!=='earth'||!state.earthOriginTile)return;
  const gen=terrainGeneration, z=state.terrainZoom, radius=state.terrainRadius;
  const live=geoFromFlight(); state.currentGeo=live;
  const jobs=[];
  const schedules=[{z,radius,detail:false},{z:z+1,radius:0,detail:true}];
  const centers=new Map();
  for(const schedule of schedules){
    const c=tileXY(live.lat,live.lon,schedule.z);
    c.x=unwrapTileX(c.x,state.earthOriginTile.x * (2 ** (schedule.z-z)),schedule.z);
    centers.set(schedule.z,{x:Math.floor(c.x),y:Math.floor(c.y)});
    for(let dy=-schedule.radius;dy<=schedule.radius;dy++)for(let dx=-schedule.radius;dx<=schedule.radius;dx++){
      const tx=Math.floor(c.x)+dx,ty=Math.floor(c.y)+dy,key=`${schedule.z}/${tx}/${ty}`;
      if(terrainTileMap.has(key)||terrainLoading.has(key))continue;
      terrainLoading.add(key);
      const job=createTerrainTile(tx,ty,schedule.z,state.earthOriginTile,state.earthTileMeters/(2 ** (schedule.z-z)),schedule.detail).then(mesh=>{
        if(gen!==terrainGeneration){disposeObject(mesh);return;}
        terrainTileMap.set(key,mesh);terrainTiles.add(mesh);
      }).finally(()=>terrainLoading.delete(key));
      jobs.push(job);
    }
  }
  terrainTileMap.forEach((mesh,key)=>{
    const [level,rawX,rawY]=key.split('/').map(Number),center=centers.get(level);
    if(!center)return;
    const dx=Math.abs(unwrapTileX(rawX,center.x,level)-center.x),dy=Math.abs(rawY-center.y);
    const keepRadius=level===z?radius+1:1;
    if(dx>keepRadius||dy>keepRadius){terrainTiles.remove(mesh);disposeObject(mesh);terrainTileMap.delete(key);}
  });
  if(initial)await Promise.allSettled(jobs);
}

function buildingTileBounds(tx, ty, z = BUILDING_ZOOM) {
  const n = 2 ** z, wrappedX = ((tx % n) + n) % n;
  const west = wrappedX / n * 360 - 180;
  const east = (wrappedX + 1) / n * 360 - 180;

  const south = THREE.MathUtils.radToDeg(Math.atan(Math.sinh(Math.PI * (1 - 2 * (ty + 1) / n))));
  return { south, west, north, east };
}

function buildingKind(tags, height) {
  const material = String(tags['building:material'] || '').toLowerCase();
  const type = String(tags.building || '').toLowerCase();
  if (material.includes('glass') || height > 80 || ['office', 'commercial', 'retail'].includes(type)) return 'glass';
  if (material.includes('brick') || ['house', 'detached', 'semidetached_house', 'terrace'].includes(type)) return 'brick';
  if (['industrial', 'warehouse', 'factory'].includes(type)) return 'industrial';
  return 'default';
}

function parseBuildingHeight(tags = {}, area = 0, seed = 0) {
  const rawHeight = parseFloat(String(tags.height || '').replace(',', '.').replace(/[^0-9.]/g, ''));
  if (Number.isFinite(rawHeight) && rawHeight > 1) return THREE.MathUtils.clamp(rawHeight, 2.5, 420);
  const levels = parseFloat(String(tags['building:levels'] || '').replace(',', '.'));
  if (Number.isFinite(levels) && levels > 0) return THREE.MathUtils.clamp(levels * 3.2, 3, 420);
  const footprintSignal = Math.sqrt(Math.max(25, area));
  const variation = 0.82 + ((Math.abs(seed * 1103515245 + 12345) % 1000) / 1000) * .36;
  return THREE.MathUtils.clamp((5.5 + footprintSignal * .055) * variation, 4.5, 28);
}

function makeBuildingTile(tx, ty, elements) {
  const groups = { default: [], glass: [], brick: [], industrial: [] };
  let accepted = 0;
  for (const way of elements) {
    if (accepted >= BUILDING_TILE_LIMIT || !way.geometry || way.geometry.length < 4) continue;
    try {
      const points = way.geometry.map(point => localFromGeo(point.lat, point.lon));
      let area = 0, cx = 0, cz = 0;
      for (let i = 0; i < points.length; i++) {
        const a = points[i], b = points[(i + 1) % points.length];
        area += a.x * b.z - b.x * a.z;
        cx += a.x; cz += a.z;
      }
      area = Math.abs(area) * .5;
      cx /= points.length; cz /= points.length;
      if (!Number.isFinite(area) || area < 12) continue;
      const shape = new THREE.Shape();
      shape.moveTo(points[0].x, -points[0].z);
      for (let i = 1; i < points.length; i++) shape.lineTo(points[i].x, -points[i].z);
      shape.closePath();
      const tags = way.tags || {};
      const height = parseBuildingHeight(tags, area, Number(way.id) || accepted);
      const geometry = new THREE.ExtrudeGeometry(shape, { depth: height, bevelEnabled: false, steps: 1 });
      geometry.rotateX(-Math.PI / 2);
      geometry.translate(0, terrainHeightAt(cx, cz) + .25, 0);
      groups[buildingKind(tags, height)].push(geometry);
      accepted++;
    } catch (error) {
      // A malformed OSM polygon should not prevent the rest of the tile rendering.
    }
  }
  const group = new THREE.Group();
  group.name = `osm-buildings-${BUILDING_ZOOM}-${tx}-${ty}`;
  for (const [kind, geometries] of Object.entries(groups)) {
    if (!geometries.length) continue;
    const merged = new THREE.BufferGeometry();
    let vertexOffset = 0;
    const positionArrays = [];
    const normalArrays = [];
    const uvArrays = [];
    const indexArrays = [];
    for (const geometry of geometries) {
      const position = geometry.getAttribute('position');
      const normal = geometry.getAttribute('normal');
      const uv = geometry.getAttribute('uv');
      positionArrays.push(position.array); if (normal) normalArrays.push(normal.array); if (uv) uvArrays.push(uv.array);
      const index = geometry.getIndex();
      if (index) indexArrays.push(Array.from(index.array, value => value + vertexOffset));
      else indexArrays.push(Array.from({ length: position.count }, (_, i) => i + vertexOffset));
      vertexOffset += position.count;
      geometry.dispose();
    }
    merged.setAttribute('position', new THREE.Float32BufferAttribute(Float32Array.from(positionArrays.flatMap(array => Array.from(array))), 3));
    if (normalArrays.length === geometries.length) merged.setAttribute('normal', new THREE.Float32BufferAttribute(Float32Array.from(normalArrays.flatMap(array => Array.from(array))), 3));
    if (uvArrays.length === geometries.length) merged.setAttribute('uv', new THREE.Float32BufferAttribute(Float32Array.from(uvArrays.flatMap(array => Array.from(array))), 2));
    merged.setIndex(indexArrays.flat());
    if (!normalArrays.length) merged.computeVertexNormals();
    merged.computeBoundingSphere();
    const mesh = new THREE.Mesh(merged, makeBuildingMaterial(kind));
    mesh.userData.buildingCount = geometries.length;
    group.add(mesh);
  }
  group.userData.buildingCount = accepted;
  group.add(makeVegetationTile(tx,ty,accepted));
  return group;
}

function parseOsmBuildingXml(xmlText) {
  const doc = new DOMParser().parseFromString(xmlText, 'application/xml');
  if (doc.querySelector('parsererror')) throw new Error('OSM XML parse failed');
  const nodes = new Map();
  doc.querySelectorAll('node').forEach(node => {
    const lat = Number(node.getAttribute('lat')), lon = Number(node.getAttribute('lon'));
    if (Number.isFinite(lat) && Number.isFinite(lon)) nodes.set(node.getAttribute('id'), { lat, lon });
  });
  const elements = [];
  doc.querySelectorAll('way').forEach(way => {
    const tags = {};
    way.querySelectorAll('tag').forEach(tag => tags[tag.getAttribute('k')] = tag.getAttribute('v') || '');
    if (!tags.building) return;
    const geometry = [];
    way.querySelectorAll('nd').forEach(nd => {
      const point = nodes.get(nd.getAttribute('ref'));
      if (point) geometry.push(point);
    });
    if (geometry.length >= 4) elements.push({ id: way.getAttribute('id'), tags, geometry });
  });
  return elements;
}

async function fetchOsmBuildingTile(bounds) {
  const url = `https://api.openstreetmap.org/api/0.6/map?bbox=${bounds.west},${bounds.south},${bounds.east},${bounds.north}`;
  const response = await fetch(url, { headers: { Accept: 'application/xml' }, cache: 'no-store' });
  if (!response.ok) throw new Error(`OSM map ${response.status}`);
  return parseOsmBuildingXml(await response.text());
}

async function fetchOverpassBuildingTile(endpoint, query) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 9000);
  try {
    const response = await fetch(endpoint, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded;charset=UTF-8', Accept: 'application/json' },
      body: `data=${encodeURIComponent(query)}`,
      signal: controller.signal,
      cache: 'no-store',
    });
    if (!response.ok) throw new Error(`Overpass ${response.status}`);
    const json = await response.json();
    return json.elements || [];
  } finally {
    clearTimeout(timeout);
  }
}

async function fetchBuildingTile(tx, ty, generation) {
  const key = `${BUILDING_ZOOM}/${tx}/${ty}`;
  const bounds = buildingTileBounds(tx, ty);
  const query = `[out:json][timeout:12];way["building"](${bounds.south},${bounds.west},${bounds.north},${bounds.east});out geom;`;
  const sources = [
    { name: 'OpenStreetMap API', load: () => fetchOsmBuildingTile(bounds) },
    ...BUILDING_ENDPOINTS.map(endpoint => ({ name: endpoint, load: () => fetchOverpassBuildingTile(endpoint, query) })),
  ];
  let lastError = null;
  for (const source of sources) {
    try {
      const elements = await source.load();
      if (generation !== buildingStreamGeneration || !state.buildings) return;
      const tile = makeBuildingTile(tx, ty, elements);
      buildingTileMap.set(key, tile); buildingsGroup.add(tile);
      return tile;
    } catch (error) {
      lastError = error;
      console.warn(`Building source ${source.name} failed for ${key}`, error);
    }
  }
  console.warn(`Building tile ${key} unavailable`, lastError);
  return null;
}
async function loadBuildings(lat, lon) {
  if (!state.buildings || state.mode !== 'earth') return;
  const center = tileXY(lat, lon, BUILDING_ZOOM);
  const cx = Math.floor(center.x), cy = Math.floor(center.y);
  state.buildingTileCenter = `${BUILDING_ZOOM}/${cx}/${cy}`;
  const jobs = [];
  for (let dy = -BUILDING_TILE_RADIUS; dy <= BUILDING_TILE_RADIUS; dy++) {
    for (let dx = -BUILDING_TILE_RADIUS; dx <= BUILDING_TILE_RADIUS; dx++) {
      const tx = cx + dx, ty = cy + dy;
      const key = `${BUILDING_ZOOM}/${tx}/${ty}`;
      if (buildingTileMap.has(key) || buildingTileLoading.has(key)) continue;
      const failedAt = buildingTileFailed.get(key);
      if (failedAt && Date.now() - failedAt < 60_000) continue;
      buildingTileLoading.add(key);
      const generation = buildingStreamGeneration;
      jobs.push(fetchBuildingTile(tx, ty, generation).then(tile => {
        if (!tile) buildingTileFailed.set(key, Date.now());
        else buildingTileFailed.delete(key);
        return tile;
      }).finally(() => buildingTileLoading.delete(key)));
    }
  }
  buildingTileMap.forEach((group, key) => {
    const [, rawX, rawY] = key.split('/').map(Number);
    const dx = Math.abs(rawX - cx), dy = Math.abs(rawY - cy);
    if (dx > BUILDING_TILE_RADIUS + 1 || dy > BUILDING_TILE_RADIUS + 1) {
      buildingsGroup.remove(group); disposeObject(group); buildingTileMap.delete(key);
    }
  });
  state.buildingCenter = { lat, lon };
  state.buildingLoading = true;
  await Promise.allSettled(jobs);
  state.buildingLoading = buildingTileLoading.size > 0;
  const visibleCount = [...buildingTileMap.values()].reduce((total, group) => total + (group.userData.buildingCount || 0), 0);
  if (visibleCount && state.buildingToastKey !== state.buildingTileCenter) {
    state.buildingToastKey = state.buildingTileCenter;
    toast(`${visibleCount.toLocaleString()} mapped 3D buildings loaded`);
  } else if (!visibleCount && jobs.length) {
    toast('Building data unavailable here; retrying nearby tiles');
  }
}
