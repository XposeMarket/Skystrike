;
  terrainTileMap.forEach(mesh=>{terrainTiles.remove(mesh);disposeObject(mesh)});
  terrainTileMap.clear(); terrainLoading.clear();
}

function unwrapTileX(x,ref,z){const n=2**z;while(x-ref>n/2)x-=n;while(x-ref<-n/2)x+=n;return x;}
function metersPerLonDegree(lat){return 111320*Math.max(.02,Math.cos(lat*Math.PI/180));}
function geoFromFlight(){
  const lat=THREE.MathUtils.clamp(state.earthOrigin.lat-flightPos.z/111320,-85.04,85.04);
  let lon=state.earthOrigin.lon+flightPos.x/metersPerLonDegree(state.earthOrigin.lat);
  lon=((lon+540)%360)-180;
  return {lat,lon};
}
function localFromGeo(lat,lon){
  let dlon=lon-state.earthOrigin.lon;if(dlon>180)dlon-=360;if(dlon<-180)dlon+=360;
  return {x:dlon*metersPerLonDegree(state.earthOrigin.lat),z:(state.earthOrigin.lat-lat)*111320};
}

function terrainHeightAt(x,z){
  for(const mesh of terrainTileMap.values()){
    const {size,seg,heights}=mesh.userData;
    const u=(x-(mesh.position.x-size/2))/size;
    const v=(z-(mesh.position.z-size/2))/size;
    if(u<0||u>1||v<0||v>1)continue;
    const gx=THREE.MathUtils.clamp(u*seg,0,seg), gy=THREE.MathUtils.clamp(v*seg,0,seg);
    const x0=Math.floor(gx),y0=Math.floor(gy),x1=Math.min(seg,x0+1),y1=Math.min(seg,y0+1);
    const fx=gx-x0,fy=gy-y0,idx=(xx,yy)=>yy*(seg+1)+xx;
    const h00=heights[idx(x0,y0)],h10=heights[idx(x1,y0)],h01=heights[idx(x0,y1)],h11=heights[idx(x1,y1)];
    return THREE.MathUtils.lerp(THREE.MathUtils.lerp(h00,h10,fx),THREE.MathUtils.lerp(h01,h11,fx),fy);
  }
  return state.terrainBase||0;
}

async function refreshTerrainTiles(initial=false){
  if(state.mode!=='earth'||!state.earthOriginTile)return;
  const gen=terrainGeneration, z=state.terrainZoom, radius=state.terrainRadius;
  const live=geoFromFlight(); state.currentGeo=live;
  const c=tileXY(live.lat,live.lon,z);c.x=unwrapTileX(c.x,state.earthOriginTile.x,z);
  const cx=Math.floor(c.x),cy=Math.floor(c.y),desired=new Set(),jobs=[];
  for(let dy=-radius;dy<=radius;dy++)for(let dx=-radius;dx<=radius;dx++){
    const tx=cx+dx,ty=cy+dy,key=`${z}/${tx}/${ty}`;desired.add(key);
    if(terrainTileMap.has(key)||terrainLoading.has(key))continue;
    terrainLoading.add(key);
    const job=createTerrainTile(tx,ty,z,state.earthOriginTile,state.earthTileMeters).then(mesh=>{
      if(gen!==terrainGeneration){disposeObject(mesh);return;}
      terrainTileMap.set(key,mesh);terrainTiles.add(mesh);
    }).finally(()=>terrainLoading.delete(key));
    jobs.push(job);
  }
  terrainTileMap.forEach((mesh,key)=>{
    const dx=Math.abs(mesh.userData.tx-cx),dy=Math.abs(mesh.userData.ty-cy);
    if(dx>radius+1||dy>radius+1){terrainTiles.remove(mesh);disposeObject(mesh);terrainTileMap.delete(key);}
  });
  if(initial)await Promise.allSettled(jobs);
}

  const mat=new THREE.SpriteMaterial({color:0xffffff,transparent:true,opacity:.2,depthWrite:false});

    const sc=700+Math.random()*1500;s.scale.set(sc,sc*.25,1);cloudGroup.add(s);
  }
  mat.dispose();
}

function geoDistanceMeters(a,b){const lat=(a.lat+b.lat)*.5*Math.PI/180,dx=(a.lon-b.lon)*111320*Math.cos(lat),dz=(a.lat-b.lat)*111320;return Math.hypot(dx,dz);}

function parseBuildingHeight(tags={}){
  const raw=parseFloat(String(tags.height||'').replace(/[^0-9.]/g,''));
  if(Number.isFinite(raw)&&raw>1)return THREE.MathUtils.clamp(raw,2,420);
  const levels=parseFloat(tags['building:levels']);
  if(Number.isFinite(levels)&&levels>0)return THREE.MathUtils.clamp(levels*3.2,3,420);
  return 9;
}

async function loadBuildings(lat,lon){
  if(!state.buildings||state.buildingLoading)return;
  state.buildingLoading=true;const gen=++buildingGeneration;
  const query=`[out:json][timeout:18];way["building"](around:1700,${lat},${lon});out geom;`;
  try{
    const r=await fetch('https://overpass-api.de/api/interpreter',{method:'POST',headers:{'Content-Type':'application/x-www-form-urlencoded;charset=UTF-8'},body:`data=${encodeURIComponent(query)}`});
    if(!r.ok)throw new Error(`Overpass ${r.status}`);const json=await r.json();if(gen!==buildingGeneration)return;
    clearGroup(buildingsGroup);
    const material=new THREE.MeshStandardMaterial({color:0xb7b2a7,roughness:.88,metalness:.02});
    let made=0;
    for(const way of json.elements||[]){
      if(made>=700||!way.geometry||way.geometry.length<4)continue;
      try{
        const pts=way.geometry.map(g=>localFromGeo(g.lat,g.lon));
        const shape=new THREE.Shape();shape.moveTo(pts[0].x,-pts[0].z);for(let i=1;i<pts.length;i++)shape.lineTo(pts[i].x,-pts[i].z);
        const h=parseBuildingHeight(way.tags),geo=new THREE.ExtrudeGeometry(shape,{depth:h,bevelEnabled:false,steps:1});geo.rotateX(-Math.PI/2);
        let cx=0,cz=0;pts.forEach(p=>{cx+=p.x;cz+=p.z});cx/=pts.length;cz/=pts.length;
        const mesh=new THREE.Mesh(geo,material);mesh.position.y=terrainHeightAt(cx,cz)+.3;mesh.userData.sharedMaterial=true;buildingsGroup.add(mesh);made++;
      }catch{}
    }
    if(!made)material.dispose();state.buildingCenter={lat,lon};if(made)toast(`${made.toLocaleString()} mapped buildings loaded`);
  }catch{state.buildingCenter={lat,lon};}
  finally{state.buildingLoading=false;}
}

async function rebaseEarth(){
  if(state.rebasing||state.mode!=='earth'||Math.hypot(flightPos.x,flightPos.z)<50000)return;
  state.rebasing=true;
  const live=geoFromFlight(),agl=Math.max(25,flightPos.y-terrainHeightAt(flightPos.x,flightPos.z));

  state.earthOriginTile=tileXY(live.lat,live.lon,state.terrainZoom);
  state.earthTileMeters=40075016.686*Math.cos(live.lat*Math.PI/180)/(2**state.terrainZoom);
  flightPos.x=0;flightPos.z=0;clearTerrain();buildingGeneration++;clearGroup(buildingsGroup);state.buildingCenter=null;
  await refreshTerrainTiles(true);state.terrainBase=terrainHeightAt(0,0);flightPos.y=Math.max(state.terrainBase+agl,35);buildClouds();loadBuildings(live.lat,live.lon);state.rebasing=false;
}
