function tuneResolution(dt){
  perfElapsed+=dt;perfFrames++;perfCooldown=Math.max(0,perfCooldown-dt);if(perfElapsed<2||perfCooldown>0)return;
  const fps=perfFrames/perfElapsed;let next=renderPixelRatio;if(fps<47)next=Math.max(PERF.minPixelRatio,renderPixelRatio-.08);else if(fps>58)next=Math.min(PERF.maxPixelRatio,renderPixelRatio+.04);
  perfElapsed=0;perfFrames=0;if(Math.abs(next-renderPixelRatio)>.001){renderPixelRatio=next;renderer.setPixelRatio(renderPixelRatio);renderer.setSize(innerWidth,innerHeight,false);perfCooldown=2.5}
}
