const AU_MILLION_KM=149.5978707;
function elementsAt(name,date=new Date()){
  const e=jplElements[name];if(!e)return null;
  const T=((date.getTime()/86400000+2440587.5)-2451545.0)/36525;
  return e[0].map((v,i)=>v+e[1][i]*T);
}
function jplEclipticPoint(name,date=new Date(),forcedE=null){
  const el=elementsAt(name,date);if(!el)return null;
  const [a,e,I,L,peri,node]=el,d=Math.PI/180;
  const omega=(peri-node)*d,O=node*d,inc=I*d;
  let E=forcedE;
  if(E==null){
    let M=((L-peri+540)%360-180)*d;E=M;
    for(let k=0;k<10;k++){const step=(E-e*Math.sin(E)-M)/(1-e*Math.cos(E));E-=step;if(Math.abs(step)<1e-10)break;}
  }
  const xp=a*(Math.cos(E)-e),yp=a*Math.sqrt(1-e*e)*Math.sin(E);
  const cw=Math.cos(omega),sw=Math.sin(omega),cO=Math.cos(O),sO=Math.sin(O),ci=Math.cos(inc),si=Math.sin(inc);
  const x=(cw*cO-sw*sO*ci)*xp+(-sw*cO-cw*sO*ci)*yp;
  const y=(cw*sO+sw*cO*ci)*xp+(-sw*sO+cw*cO*ci)*yp;
  const z=(sw*si)*xp+(cw*si)*yp;
  return new THREE.Vector3(x,z,y); // ecliptic z becomes scene vertical; ecliptic y becomes scene depth.
}
